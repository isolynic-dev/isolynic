Building this as a real Next.js App Router feature: types, a Firestore real-time hook, an actions layer with idempotency/dedup logic, and the UI components, wired to a `/recovery` page.

```tsx
// types/opportunity.ts

export type RiskState =
  | 'NEW_RISK'
  | 'RECOVERY_RECOMMENDED'
  | 'OWNER_APPROVED'
  | 'RECOVERY_SENT'
  | 'WAITING'
  | 'CUSTOMER_RESPONDED'
  | 'PROGRESSING'
  | 'BOOKED'
  | 'WON'
  | 'LOST'
  | 'HANDLED';

export type PriorityLevel = 'high' | 'worth_checking' | 'low';

export type ChannelType = 'phone' | 'whatsapp' | 'website' | 'sms' | 'email';

export type OwnerActionType =
  | 'recover'
  | 'handle_myself'
  | 'ignore'
  | 'not_a_customer'
  | null;

export interface EvidenceItem {
  id: string;
  event_type: string; // e.g. "missed_call", "quote_sent", "no_response"
  timestamp: number; // epoch ms
  channel: ChannelType;
  summary: string;
  relevance: number; // 0-1, internal only, never rendered directly
}

export interface Opportunity {
  id: string;
  businessId: string;
  customer_id: string;
  customer_name: string;
  source_channels: ChannelType[];

  first_contact_at: number;
  latest_activity_at: number;

  intent_summary: string; // "Asked for a repair appointment yesterday."
  why_now: string; // "You sent a quote 3 days ago and haven't heard back."
  recommendation: string; // "A short follow-up could bring this customer back."

  priority: PriorityLevel;
  priority_score: number; // internal, never rendered

  value_estimate?: number | null; // optional, internal-derived

  risk_state: RiskState;
  status: 'active' | 'waiting' | 'responded' | 'resolved' | 'lost' | 'handled';

  needs_owner: boolean; // true => "Needs you"
  blocked_reason?: string | null; // e.g. "missing pricing info"

  owner_action: OwnerActionType;
  last_recovery_action_id?: string | null;
  follow_up_count: number;
  next_review_at?: number | null;

  evidence: EvidenceItem[];

  outcome?: {
    type: 'booked' | 'won' | 'lost' | 'handled' | null;
    note?: string;
    at?: number;
  } | null;

  presence?: {
    handled_by_uid: string;
    handled_by_name: string;
  } | null;

  recovery_preview?: {
    message: string;
    requires_confirmation: boolean;
  } | null;

  createdAt: number;
  updatedAt: number;
}

export interface RecoveredEntry {
  id: string;
  customer_name: string;
  outcome_text: string; // "Customer replied after follow-up."
  resolved_at: number;
}

export type QueueFilter = 'all' | 'needs_you' | 'handled';
```

## 3. `hooks/useRecoveryQueue.ts`

Real-time Firestore subscription with the priority sort, filtering, and offline/error handling described in the spec (secs 8–11, 24, 50–51).

```tsx
// hooks/useRecoveryQueue.ts
'use client';

import { useEffect, useMemo, useState, useCallback, useRef } from 'react';
import {
  collection,
  query,
  where,
  orderBy,
  onSnapshot,
  limit,
  Unsubscribe,
} from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { Opportunity, QueueFilter, RecoveredEntry } from '@/types/opportunity';

const ACTIVE_RISK_STATES = [
  'NEW_RISK',
  'RECOVERY_RECOMMENDED',
  'OWNER_APPROVED',
  'RECOVERY_SENT',
  'WAITING',
  'CUSTOMER_RESPONDED',
  'PROGRESSING',
];

interface UseRecoveryQueueResult {
  opportunities: Opportunity[];
  recentlyRecovered: RecoveredEntry[];
  loading: boolean;
  isStale: boolean; // showing cached data, couldn't refresh
  error: string | null;
  isOffline: boolean;
  retry: () => void;
  needsYouCount: number;
  autoHandledCount: number;
}

const CACHE_KEY = (businessId: string) => `isolynic:recoveryQueue:${businessId}`;

export function useRecoveryQueue(
  businessId: string,
  filter: QueueFilter
): UseRecoveryQueueResult {
  const [opportunities, setOpportunities] = useState<Opportunity[]>([]);
  const [recentlyRecovered, setRecentlyRecovered] = useState<RecoveredEntry[]>([]);
  const [loading, setLoading] = useState(true);
  const [isStale, setIsStale] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [isOffline, setIsOffline] = useState(
    typeof navigator !== 'undefined' ? !navigator.onLine : false
  );
  const [retryTick, setRetryTick] = useState(0);
  const unsubRef = useRef<Unsubscribe | null>(null);

  // Hydrate from local cache immediately so the screen never opens blank
  useEffect(() => {
    if (typeof window === 'undefined') return;
    try {
      const cached = window.localStorage.getItem(CACHE_KEY(businessId));
      if (cached) {
        const parsed = JSON.parse(cached) as Opportunity[];
        setOpportunities(parsed);
        setLoading(false);
      }
    } catch {
      // ignore corrupt cache
    }
  }, [businessId]);

  useEffect(() => {
    const goOnline = () => setIsOffline(false);
    const goOffline = () => setIsOffline(true);
    window.addEventListener('online', goOnline);
    window.addEventListener('offline', goOffline);
    return () => {
      window.removeEventListener('online', goOnline);
      window.removeEventListener('offline', goOffline);
    };
  }, []);

  useEffect(() => {
    if (!businessId) return;
    setError(null);

    const opportunitiesQuery = query(
      collection(db, 'businesses', businessId, 'opportunities'),
      where('risk_state', 'in', ACTIVE_RISK_STATES),
      orderBy('priority_score', 'desc'),
      limit(100)
    );

    if (unsubRef.current) unsubRef.current();

    unsubRef.current = onSnapshot(
      opportunitiesQuery,
      (snapshot) => {
        const items = snapshot.docs.map((d) => ({ id: d.id, ...d.data() } as Opportunity));
        setOpportunities(items);
        setLoading(false);
        setIsStale(false);
        setError(null);
        try {
          window.localStorage.setItem(CACHE_KEY(businessId), JSON.stringify(items));
        } catch {
          // storage may be full/unavailable — non-fatal
        }
      },
      (err) => {
        console.error('[RecoveryQueue] snapshot error', err);
        setLoading(false);
        setIsStale(true);
        setError(
          "We couldn't check for new customer activity. Your last results are still here."
        );
      }
    );

    return () => {
      unsubRef.current?.();
    };
  }, [businessId, retryTick]);

  // Recently recovered — small, separate, capped feed
  useEffect(() => {
    if (!businessId) return;
    const recoveredQuery = query(
      collection(db, 'businesses', businessId, 'recoveredFeed'),
      orderBy('resolved_at', 'desc'),
      limit(5)
    );
    const unsub = onSnapshot(recoveredQuery, (snapshot) => {
      setRecentlyRecovered(
        snapshot.docs.map((d) => ({ id: d.id, ...d.data() } as RecoveredEntry))
      );
    });
    return () => unsub();
  }, [businessId]);

  const retry = useCallback(() => setRetryTick((t) => t + 1), []);

  const filtered = useMemo(() => {
    switch (filter) {
      case 'needs_you':
        return opportunities.filter((o) => o.needs_you_computed ?? o.needs_owner);
      case 'handled':
        return opportunities.filter((o) =>
          ['RECOVERY_SENT', 'WAITING', 'HANDLED'].includes(o.risk_state)
        );
      default:
        return opportunities;
    }
  }, [opportunities, filter]);

  const needsYouCount = useMemo(
    () => opportunities.filter((o) => o.needs_owner).length,
    [opportunities]
  );
  const autoHandledCount = opportunities.length - needsYouCount;

  return {
    opportunities: filtered,
    recentlyRecovered,
    loading,
    isStale,
    error,
    isOffline,
    retry,
    needsYouCount,
    autoHandledCount,
  };
}
```

Note: I referenced `needs_you_computed` defensively in case a server-computed field exists — falls back to `needs_owner`. Fine as-is.

## 4. `hooks/useRecoveryActions.ts`

This is where sec. 34 (idempotency/dedup), 17–20 (recover/handle/ignore/not-a-customer), and 62 (undo) live. Actions go through Firebase Cloud Functions rather than direct client writes, since "recover" triggers real customer-facing messaging — that must be server-authorized, not client-trusted.

```tsx
// hooks/useRecoveryActions.ts
'use client';

import { useState, useCallback, useRef } from 'react';
import { httpsCallable } from 'firebase/functions';
import { functions } from '@/lib/firebase';
import { v4 as uuidv4 } from 'uuid';

type ActionName = 'recover' | 'handle_myself' | 'ignore' | 'not_a_customer' | 'undo';

interface ActionState {
  pendingIds: Set<string>;
  errorByOpportunity: Record<string, string | undefined>;
}

interface RecoverPayload {
  opportunityId: string;
  businessId: string;
  editedMessage?: string; // if owner used the "Send / Edit" preview (sec 39)
}

export function useRecoveryActions(businessId: string) {
  const [state, setState] = useState<ActionState>({
    pendingIds: new Set(),
    errorByOpportunity: {},
  });

  // idempotency keys are generated once per logical user action and reused
  // on retry so duplicate network calls never create duplicate outreach.
  const idempotencyKeys = useRef<Map<string, string>>(new Map());

  const getIdempotencyKey = (opportunityId: string, action: ActionName) => {
    const mapKey = `${action}:${opportunityId}`;
    if (!idempotencyKeys.current.has(mapKey)) {
      idempotencyKeys.current.set(mapKey, uuidv4());
    }
    return idempotencyKeys.current.get(mapKey)!;
  };

  const setPending = (id: string, pending: boolean) => {
    setState((prev) => {
      const next = new Set(prev.pendingIds);
      pending ? next.add(id) : next.delete(id);
      return { ...prev, pendingIds: next };
    });
  };

  const setError = (id: string, message: string | undefined) => {
    setState((prev) => ({
      ...prev,
      errorByOpportunity: { ...prev.errorByOpportunity, [id]: message },
    }));
  };

  const callAction = useCallback(
    async (
      action: ActionName,
      opportunityId: string,
      extra?: Record<string, unknown>
    ) => {
      setPending(opportunityId, true);
      setError(opportunityId, undefined);
      const idempotencyKey = getIdempotencyKey(opportunityId, action);
      try {
        const fn = httpsCallable(functions, `recoveryQueue-${action}`);
        await fn({
          businessId,
          opportunityId,
          idempotencyKey,
          ...extra,
        });
        // On success, clear the stored key so a genuinely new future
        // action (e.g. a second later recovery) gets a fresh one.
        idempotencyKeys.current.delete(`${action}:${opportunityId}`);
        return { ok: true as const };
      } catch (err) {
        console.error(`[RecoveryActions] ${action} failed`, err);
        setError(
          opportunityId,
          action === 'recover'
            ? "We couldn't send that message."
            : "That didn't go through."
        );
        return { ok: false as const };
      } finally {
        setPending(opportunityId, false);
      }
    },
    [businessId]
  );

  const recover = useCallback(
    (payload: RecoverPayload) =>
      callAction('recover', payload.opportunityId, {
        editedMessage: payload.editedMessage,
      }),
    [callAction]
  );

  const handleMyself = useCallback(
    (opportunityId: string) => callAction('handle_myself', opportunityId),
    [callAction]
  );

  const ignore = useCallback(
    (opportunityId: string) => callAction('ignore', opportunityId),
    [callAction]
  );

  const notACustomer = useCallback(
    (opportunityId: string) => callAction('not_a_customer', opportunityId),
    [callAction]
  );

  const undo = useCallback(
    (opportunityId: string, priorAction: ActionName) =>
      callAction('undo', opportunityId, { priorAction }),
    [callAction]
  );

  return {
    recover,
    handleMyself,
    ignore,
    notACustomer,
    undo,
    isPending: (id: string) => state.pendingIds.has(id),
    errorFor: (id: string) => state.errorByOpportunity[id],
  };
}
```

## 7. `components/recovery/OpportunityCard.tsx`

The core UI element (sec. 12–20, 31–33, 39–41, 62, 71–74).

```tsx
// components/recovery/OpportunityCard.tsx
'use client';

import { useState } from 'react';
import { Opportunity } from '@/types/opportunity';
import { PriorityBadge } from './PriorityBadge';
import { ChannelIcons } from './ChannelIcons';
import { useRecoveryActions } from '@/hooks/useRecoveryActions';
import { AlertCircle, Loader2, ChevronRight } from 'lucide-react';
import { useRouter } from 'next/navigation';

interface OpportunityCardProps {
  opportunity: Opportunity;
  businessId: string;
}

function timeAgo(ts: number): string {
  const diffMs = Date.now() - ts;
  const mins = Math.round(diffMs / 60000);
  if (mins < 1) return 'just now';
  if (mins < 60) return `${mins} minute${mins === 1 ? '' : 's'} ago`;
  const hrs = Math.round(mins / 60);
  if (hrs < 24) return `${hrs} hour${hrs === 1 ? '' : 's'} ago`;
  const days = Math.round(hrs / 24);
  return `${days} day${days === 1 ? '' : 's'} ago`;
}

export function OpportunityCard({ opportunity, businessId }: OpportunityCardProps) {
  const router = useRouter();
  const actions = useRecoveryActions(businessId);
  const [showPreview, setShowPreview] = useState(false);
  const [editedMessage, setEditedMessage] = useState(
    opportunity.recovery_preview?.message ?? ''
  );
  const [dismissed, setDismissed] = useState<null | 'ignore' | 'not_a_customer'>(null);

  const pending = actions.isPending(opportunity.id);
  const actionError = actions.errorFor(opportunity.id);

  const goToConversation = () => router.push(`/conversations/${opportunity.customer_id}`);

  const handleRecoverClick = async () => {
    if (opportunity.recovery_preview?.requires_confirmation) {
      setShowPreview(true);
      return;
    }
    await actions.recover({ opportunityId: opportunity.id, businessId });
  };

  const confirmSend = async () => {
    setShowPreview(false);
    await actions.recover({
      opportunityId: opportunity.id,
      businessId,
      editedMessage,
    });
  };

  const handleIgnore = async () => {
    const result = await actions.ignore(opportunity.id);
    if (result.ok) setDismissed('ignore');
  };

  const handleNotACustomer = async () => {
    const result = await actions.notACustomer(opportunity.id);
    if (result.ok) setDismissed('not_a_customer');
  };

  const handleUndo = async () => {
    if (!dismissed) return;
    const result = await actions.undo(opportunity.id, dismissed as any);
    if (result.ok) setDismissed(null);
  };

  // --- Dismissed (undo) state ---
  if (dismissed) {
    return (
      <div
        role="status"
        className="flex items-center justify-between rounded-xl border border-slate-200 bg-slate-50 px-4 py-3 text-sm text-slate-500"
      >
        <span>
          {dismissed === 'ignore' ? 'Marked as ignored.' : 'Marked as not a customer.'}
        </span>
        <button
          onClick={handleUndo}
          className="font-medium text-slate-700 underline-offset-2 hover:underline"
        >
          Undo
        </button>
      </div>
    );
  }

  // --- Blocked / needs-info state (sec 31–32, 89) ---
  if (opportunity.needs_owner && opportunity.blocked_reason) {
    return (
      <article
        aria-label={`${opportunity.customer_name}. Needs you. ${opportunity.blocked_reason}`}
        className="rounded-xl border border-amber-200 bg-amber-50/60 p-4 sm:p-5"
      >
        <div className="flex items-start justify-between gap-3">
          <div>
            <h3 className="font-semibold text-slate-900">{opportunity.customer_name} needs you</h3>
            <p className="mt-1 text-sm text-slate-600">{opportunity.blocked_reason}</p>
          </div>
          <ChannelIcons channels={opportunity.source_channels} />
        </div>
        <button
          onClick={goToConversation}
          className="mt-3 inline-flex items-center gap-1 rounded-lg bg-slate-900 px-3.5 py-2 text-sm font-medium text-white hover:bg-slate-800 focus:outline-none focus-visible:ring-2 focus-visible:ring-slate-900 focus-visible:ring-offset-2"
        >
          Handle myself
        </button>
      </article>
    );
  }

  // --- Already recovering / waiting state (sec 71) ---
  const isWaiting = opportunity.risk_state === 'RECOVERY_SENT' || opportunity.risk_state === 'WAITING';
  const customerResponded = opportunity.risk_state === 'CUSTOMER_RESPONDED';

  const srLabel = `${opportunity.customer_name}. ${
    opportunity.priority === 'high' ? 'High priority.' : opportunity.priority === 'worth_checking' ? 'Worth checking.' : 'Low priority.'
  } ${opportunity.why_now} Isolynic recommends: ${opportunity.recommendation}.`;

  return (
    <article
      aria-label={srLabel}
      className="rounded-xl border border-slate-200 bg-white p-4 shadow-sm transition-shadow hover:shadow-md sm:p-5"
    >
      <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
        <div className="min-w-0 flex-1">
          <div className="flex flex-wrap items-center gap-2">
            <h3 className="font-semibold text-slate-900">{opportunity.customer_name}</h3>
            <PriorityBadge level={opportunity.priority} />
            <ChannelIcons channels={opportunity.source_channels} />
          </div>

          <p className="mt-1.5 text-sm text-slate-700">{opportunity.intent_summary}</p>

          {customerResponded ? (
            <p className="mt-1 text-sm font-medium text-emerald-700">
              {opportunity.customer_name.split(' ')[0]} replied
            </p>
          ) : isWaiting ? (
            <p className="mt-1 text-sm text-slate-500">
              Follow-up sent {timeAgo(opportunity.updatedAt)} · Waiting for a reply
            </p>
          ) : (
            <>
              <p className="mt-2 text-sm text-slate-500">
                <span className="font-medium text-slate-600">Why now: </span>
                {opportunity.why_now}
              </p>
              <p className="mt-1 text-sm text-slate-500">
                <span className="font-medium text-slate-600">Isolynic recommends: </span>
                {opportunity.recommendation}
              </p>
            </>
          )}

          {actionError && (
            <div className="mt-2 flex items-center gap-1.5 text-sm text-rose-600">
              <AlertCircle className="h-4 w-4" aria-hidden="true" />
              <span>{actionError}</span>
            </div>
          )}
        </div>

        <div className="flex shrink-0 flex-row gap-2 sm:flex-col sm:items-stretch">
          {!isWaiting && !customerResponded && (
            <button
              onClick={handleRecoverClick}
              disabled={pending}
              className="inline-flex items-center justify-center gap-1.5 rounded-lg bg-slate-900 px-4 py-2 text-sm font-medium text-white hover:bg-slate-800 disabled:opacity-60 focus:outline-none focus-visible:ring-2 focus-visible:ring-slate-900 focus-visible:ring-offset-2"
              aria-label={`Recover ${opportunity.customer_name}`}
            >
              {pending ? (
                <>
                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
                  Recovering…
                </>
              ) : (
                'Recover'
              )}
            </button>
          )}

          {customerResponded && (
            <button
              onClick={goToConversation}
              className="inline-flex items-center justify-center gap-1 rounded-lg bg-slate-900 px-4 py-2 text-sm font-medium text-white hover:bg-slate-800"
            >
              Continue <ChevronRight className="h-4 w-4" aria-hidden="true" />
            </button>
          )}

          <button
            onClick={goToConversation}
            className="rounded-lg border border-slate-200 px-4 py-2 text-sm font-medium text-slate-700 hover:bg-slate-50 focus:outline-none focus-visible:ring-2 focus-visible:ring-slate-300"
          >
            Handle myself
          </button>

          {!isWaiting && !customerResponded && (
            <div className="flex gap-2 sm:hidden">
              <button
                onClick={handleNotACustomer}
                className="flex-1 rounded-lg px-3 py-1.5 text-xs text-slate-400 hover:text-slate-600"
              >
                Not a customer
              </button>
              <button
                onClick={handleIgnore}
                className="flex-1 rounded-lg px-3 py-1.5 text-xs text-slate-400 hover:text-slate-600"
              >
                Ignore
              </button>
            </div>
          )}
        </div>
      </div>

      {!isWaiting && !customerResponded && (
        <div className="mt-3 hidden gap-4 border-t border-slate-100 pt-2.5 sm:flex">
          <button
            onClick={handleNotACustomer}
            className="text-xs text-slate-400 hover:text-slate-600"
          >
            Not a customer
          </button>
          <button
            onClick={handleIgnore}
            className="text-xs text-slate-400 hover:text-slate-600"
          >
            Ignore
          </button>
        </div>
      )}

      {showPreview && (
        <div
          role="dialog"
          aria-modal="true"
          aria-label={`Confirm recovery message to ${opportunity.customer_name}`}
          className="mt-4 rounded-lg border border-slate-200 bg-slate-50 p-3.5"
        >
          <p className="mb-2 text-xs font-medium text-slate-500">Isolynic would send:</p>
          <textarea
            value={editedMessage}
            onChange={(e) => setEditedMessage(e.target.value)}
            rows={3}
```

### Notes on backend responsibilities (Cloud Functions, not shown as code since these are server-side, not TSX)

- `recoveryQueue-recover`, `-handle_myself`, `-ignore`, `-not_a_customer`, `-undo` should each: verify the caller owns `businessId`, look up the opportunity by `idempotencyKey` in a `recoveryActions` collection first (write-once), and only then mutate state — this is what actually enforces sec. 34's dedup guarantee; the client-side key alone isn't sufficient.
- Priority score, evidence, and channel merging (secs 36, 42, 76) belong in a background function that writes `priority_score`/`priority` onto the opportunity doc — never computed client-side, since the formula must stay invisible per sec. 14/42.
- The `Recover` action should never mark `message_sent = recovery_success` (sec. 22) — the function transitions `risk_state` to `RECOVERY_SENT`/`WAITING`, and only a later customer event moves it to `CUSTOMER_RESPONDED` or an outcome.

## 5. `functions/src/lib/idempotency.ts`

The core of sec. 34 — every recovery-affecting action must be idempotent regardless of retries, duplicate webhooks, or multiple simultaneous callers.

```typescript
// functions/src/lib/idempotency.ts
import { db, FieldValue } from './admin';
import { HttpsError } from 'firebase-functions/v2/https';
import { Firestore, Transaction } from 'firebase-admin/firestore';

const ACTIONS_COLLECTION = 'recoveryActions';

/**
 * Reserves an idempotency key inside the SAME transaction as the state
 * mutation it protects. If the key was already used, throws a benign
 * "already-exists" style error the caller can treat as a no-op success,
 * rather than re-running the side effect (e.g. sending a message twice).
 */
export async function reserveIdempotencyKey(
  tx: Transaction,
  params: {
    idempotencyKey: string;
    action: string;
    opportunityId: string;
    businessId: string;
    uid: string;
  }
): Promise<{ alreadyProcessed: boolean; priorResult?: unknown }> {
  const ref = db.collection(ACTIONS_COLLECTION).doc(params.idempotencyKey);
  const existing = await tx.get(ref);

  if (existing.exists) {
    const data = existing.data()!;
    return { alreadyProcessed: true, priorResult: data.result };
  }

  tx.set(ref, {
    idempotencyKey: params.idempotencyKey,
    action: params.action,
    opportunityId: params.opportunityId,
    businessId: params.businessId,
    uid: params.uid,
    status: 'pending',
    createdAt: Date.now(),
  });

  return { alreadyProcessed: false };
}

export async function finalizeIdempotencyKey(
  idempotencyKey: string,
  result: { status: 'succeeded' | 'failed'; result?: unknown; error?: string }
): Promise<void> {
  await db
    .collection(ACTIONS_COLLECTION)
    .doc(idempotencyKey)
    .set(
      {
        status: result.status,
        result: result.result ?? null,
        error: result.error ?? null,
        finalizedAt: Date.now(),
      },
      { merge: true }
    );
}

/** Guards against a stuck "pending" key from a crashed invocation blocking retries forever. */
export function isStalePending(createdAt: number, maxAgeMs = 60_000): boolean {
  return Date.now() - createdAt > maxAgeMs;
}
```

```typescript
// functions/src/lib/antiSpam.ts
import { OpportunityDoc } from '../types';

export interface AntiSpamResult {
  allowed: boolean;
  reason?: string;
}

const DEFAULT_COOLDOWN_MS = 24 * 60 * 60 * 1000; // 24h between automated follow-ups
const DEFAULT_MAX_FOLLOW_UPS = 3;

export function evaluateRecoveryEligibility(opp: OpportunityDoc): AntiSpamResult {
  if (opp.customer_opted_out) {
    return { allowed: false, reason: 'This customer asked not to be contacted again.' };
  }

  if (opp.customer_rejected_at) {
    return { allowed: false, reason: 'This customer already declined.' };
  }

  if (opp.risk_state === 'CUSTOMER_RESPONDED' || opp.risk_state === 'PROGRESSING') {
    return { allowed: false, reason: 'This customer already responded — no follow-up needed.' };
  }

  if (['BOOKED', 'WON', 'LOST', 'HANDLED'].includes(opp.risk_state)) {
    return { allowed: false, reason: 'This opportunity is already resolved.' };
  }

  const maxFollowUps = opp.max_follow_ups ?? DEFAULT_MAX_FOLLOW_UPS;
  if ((opp.follow_up_count ?? 0) >= maxFollowUps) {
    return { allowed: false, reason: "We've already followed up enough times." };
  }

  if (opp.cooldown_until && Date.now() < opp.cooldown_until) {
    return { allowed: false, reason: 'No follow-up needed right now.' };
  }

  if (opp.presence?.handled_by_uid) {
    return {
      allowed: false,
      reason: `${opp.presence.handled_by_name} is already handling this customer.`,
    };
  }

  return { allowed: true };
}

export function nextCooldownTimestamp(): number {
  return Date.now() + DEFAULT_COOLDOWN_MS;
}
```

```typescript
// functions/src/recoveryQueue/recover.ts
import { onCall, HttpsError } from 'firebase-functions/v2/https';
import { db, FieldValue } from '../lib/admin';
import { requireAuth, badRequest, notFound, failedPrecondition } from '../lib/errors';
import { assertBusinessMember } from '../lib/authz';
import { reserveIdempotencyKey, finalizeIdempotencyKey } from '../lib/idempotency';
import { evaluateRecoveryEligibility, nextCooldownTimestamp } from '../lib/antiSpam';
import { composeRecoveryMessage, sendRecoveryMessage } from '../lib/messaging';
import { OpportunityDoc } from '../types';

interface RecoverPayload {
  businessId: string;
  opportunityId: string;
  idempotencyKey: string;
  editedMessage?: string;
}

export const recover = onCall<RecoverPayload>(
  { region: 'us-central1', cors: true },
  async (request) => {
    const uid = request.auth?.uid;
    requireAuth(uid);

    const { businessId, opportunityId, idempotencyKey, editedMessage } = request.data ?? {};
    if (!businessId || !opportunityId || !idempotencyKey) {
      badRequest('businessId, opportunityId, and idempotencyKey are required.');
    }

    await assertBusinessMember(businessId, uid);

    const oppRef = db
      .collection('businesses')
      .doc(businessId)
      .collection('opportunities')
      .doc(opportunityId);

    // Step 1: reserve the idempotency key and read+validate opportunity
    // state inside one transaction, so a duplicate call (retry, double
    // webhook, double click) can never pass the eligibility check twice.
    const reservation = await db.runTransaction(async (tx) => {
      const oppSnap = await tx.get(oppRef);
      if (!oppSnap.exists) notFound();

      const opp = oppSnap.data() as OpportunityDoc;
      if (opp.businessId !== businessId) notFound();

      const idemp = await reserveIdempotencyKey(tx, {
        idempotencyKey,
        action: 'recover',
        opportunityId,
        businessId,
        uid,
      });

      if (idemp.alreadyProcessed) {
        return { skip: true as const, opp };
      }

      const eligibility = evaluateRecoveryEligibility(opp);
      if (!eligibility.allowed) {
        // Mark the reserved key failed so it doesn't dangle as "pending".
        tx.set(
          db.collection('recoveryActions').doc(idempotencyKey),
          { status: 'failed', error: eligibility.reason, finalizedAt: Date.now() },
          { merge: true }
        );
        return { skip: false as const, blocked: eligibility.reason, opp };
      }

      // Optimistically mark as owner-approved / sending so concurrent
      // calls (e.g. two tabs) see this opportunity is already in flight.
      tx.update(oppRef, {
        risk_state: 'OWNER_APPROVED',
        owner_action: 'recover',
        updatedAt: Date.now(),
      });

      return { skip: false as const, opp };
    });

    if (reservation.skip) {
      // Already handled by a prior identical call — return the same
      // outcome without sending anything again.
      return { ok: true, alreadyProcessed: true };
    }

    if (reservation.blocked) {
      return { ok: false, reason: reservation.blocked };
    }

    const opp = reservation.opp;
    const channel = opp.source_channels?.[0] ?? 'sms';

    try {
      const message =
        editedMessage?.trim() ||
        (await composeRecoveryMessage({
          customerName: opp.customer_name,
          intentSummary: opp.intent_summary,
          whyNow: opp.why_now,
        }));

      const sendResult = await sendRecoveryMessage({
        businessId,
        customerId: opp.customer_id,
        channel,
        message,
      });

      if (!sendResult.delivered) {
        await oppRef.update({
          risk_state: 'RECOVERY_RECOMMENDED', // roll back to pre-send state
          updatedAt: Date.now(),
        });
        await finalizeIdempotencyKey(idempotencyKey, {
          status: 'failed',
          error: sendResult.error ?? 'send_failed',
        });
        throw new HttpsError('unavailable', "We couldn't send that message.");
      }

      // Action success (sec 22) — NOT opportunity success. risk_state
      // moves to RECOVERY_SENT/WAITING; only a later customer event
      // advances it to CUSTOMER_RESPONDED/PROGRESSING/BOOKED etc.
      const now = Date.now();
      await oppRef.update({
        risk_state: 'WAITING',
        status: 'waiting',
        last_recovery_action_id: idempotencyKey,
        follow_up_count: FieldValue.increment(1),
        cooldown_until: nextCooldownTimestamp(),
        latest_activity_at: now,
        updatedAt: now,
        evidence: FieldValue.arrayUnion({
          id: idempotencyKey,
          event_type: 'recovery_message_sent',
          timestamp: now,
          channel,
          summary: 'Follow-up sent.',
          relevance: 1,
        }),
      });

      await finalizeIdempotencyKey(idempotencyKey, {
        status: 'succeeded',
        result: { providerMessageId: sendResult.providerMessageId },
      });

      return { ok: true, alreadyProcessed: false };
    } catch (err) {
      await finalizeIdempotencyKey(idempotencyKey, {
        status: 'failed',
        error: err instanceof Error ? err.message : 'unknown_error',
      });
      if (err instanceof HttpsError) throw err;
      throw new HttpsError('internal', "We couldn't send that message.");
    }
  }
);
```

```typescript
          .collection('businesses')
          .doc(businessId)
          .collection('learningSignals')
          .doc(category);

          signalRef,

        );
      }

function classifyForLearning(opp: OpportunityDoc): string | undefined {
  // Simplistic bucket by first evidence event_type + channel; a real
  // implementation would use the same feature set as the scoring model.
  const firstEvidence = opp.evidence?.[0];
  if (!firstEvidence) return undefined;
  return `${firstEvidence.channel}:${firstEvidence.event_type}`;
```

## 12. `functions/src/recoveryQueue/undo.ts`

Sec. 62 — restores the pre-dismiss snapshot within a short window.

```typescript
// functions/src/recoveryQueue/undo.ts
import { onCall } from 'firebase-functions/v2/https';
import { db } from '../lib/admin';
import { requireAuth, badRequest, notFound, failedPrecondition } from '../lib/errors';
import { assertBusinessMember } from '../lib/authz';
import { reserveIdempotencyKey, finalizeIdempotencyKey } from '../lib/idempotency';
import { CallablePayload, OpportunityDoc } from '../types';

const UNDO_WINDOW_MS = 10_000;

interface UndoPayload extends CallablePayload {
  priorAction: 'ignore' | 'not_a_customer';
}

export const undo = onCall<UndoPayload>(
  { region: 'us-central1', cors: true },
  async (request) => {
    const uid = request.auth?.uid;
    requireAuth(uid);

    const { businessId, opportunityId, idempotencyKey, priorAction } = request.data ?? {};
    if (!businessId || !opportunityId || !idempotencyKey || !priorAction) {
      badRequest('businessId, opportunityId, idempotencyKey, and priorAction are required.');
    }

    await assertBusinessMember(businessId, uid);

    const oppRef = db
      .collection('businesses')
      .doc(businessId)
      .collection('opportunities')
      .doc(opportunityId);

    await db.runTransaction(async (tx) => {
      const oppSnap = await tx.get(oppRef);
      if (!oppSnap.exists) notFound();
      const opp = oppSnap.data() as OpportunityDoc;

      const idemp = await reserveIdempotencyKey(tx, {
        idempotencyKey,
        action: 'undo',
        opportunityId,
        businessId,
        uid,
      });
      if (idemp.alreadyProcessed) return;

      if (opp.owner_action !== priorAction) {
        failedPrecondition('This action can no longer be undone.');
      }
      if (Date.now() - opp.updatedAt > UNDO_WINDOW_MS) {
        failedPrecondition('The window to undo this action has passed.');
      }
      if (!opp.pre_dismiss_snapshot) {
        failedPrecondition('Nothing to undo.');
      }

      tx.update(oppRef, {
        ...opp.pre_dismiss_snapshot,
        pre_dismiss_snapshot: null,
        updatedAt: Date.now(),
      });
    });

    await finalizeIdempotencyKey(idempotencyKey, { status: 'succeeded' });

    return { ok: true };
  }
);
```
